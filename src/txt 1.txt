// 1. Context Setup
const CurrencyContext = createContext();
export const useCurrencyContext = () => useContext(CurrencyContext);

// 2. Custom Hook for Dynamic Currency Rates
const useDynamicCurrency = (db, currency, isAuthReady, appId) => {
   const [conversionRate, setConversionRate] = useState(1.0); // Default to 1.0 (USD)
   const [isLoading, setIsLoading] = useState(false);
   const [error, setError] = useState(null);

   // This effect handles real-time rate fetching whenever the currency or auth status changes.
   useEffect(() => {
      if (!db || !isAuthReady || !appId) {
         // Cannot run query until Firebase is initialized and auth is ready
         return;
      }

      const fetchRates = async () => {
         setIsLoading(true);
         setError(null);

         // The appId variable must be used to construct the public collection path
         const collectionPath = `artifacts/${appId}/public/data/currency_rates`;

         // Note: We use the currency code as the document ID for efficiency
         const rateDocRef = doc(db, collectionPath, currency);

         // Set up real-time listener
         const unsubscribe = onSnapshot(
            rateDocRef,
            (docSnap) => {
               if (docSnap.exists()) {
                  const data = docSnap.data();
                  const rate = data.rate || 1.0;
                  setConversionRate(rate);
                  setError(null);
               } else {
                  // Fallback if rate document doesn't exist (e.g., EUR/USD/GBP/JPY is missing)
                  console.warn(
                     `Currency rate document for ${currency} not found. Defaulting to 1.0.`
                  );
                  setConversionRate(1.0);
                  setError(`Rate for ${currency} not available.`);
               }
               setIsLoading(false);
            },
            (err) => {
               // Handle Firebase errors
               console.error('Firestore onSnapshot failed:', err);
               setError(`Failed to fetch rates: ${err.message}`);
               setConversionRate(1.0);
               setIsLoading(false);
            }
         );

         // Cleanup function for the real-time listener
         return () => unsubscribe();
      };

      // Fetch rates only if the target currency is not USD (since USD rate is always 1.0)
      if (currency !== 'USD') {
         fetchRates();
      } else {
         setConversionRate(1.0);
         setIsLoading(false);
         setError(null);
      }
   }, [db, currency, isAuthReady, appId]); // Added appId to dependencies

   return { conversionRate, isLoading, error };
};

const App = () => {
   // Environment Variable Access - Consolidated and simplified
   const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';
   const firebaseConfig =
      typeof __firebase_config !== 'undefined'
         ? JSON.parse(__firebase_config)
         : {};
   const initialAuthToken =
      typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;

   const [db, setDb] = useState(null);
   const [auth, setAuth] = useState(null);
   const [userId, setUserId] = useState(null);
   const [isAuthReady, setIsAuthReady] = useState(false);
   const [currentPath, setCurrentPath] = useState('/');
   const [currency, setCurrency] = useState('USD');

   // Pass appId to the hook that needs it for path construction
   const { conversionRate, isLoading, error } = useDynamicCurrency(
      db,
      currency,
      isAuthReady,
      appId
   );

   // 1. Firebase Initialization and Authentication
   useEffect(() => {
      let authInstance;

      try {
         if (Object.keys(firebaseConfig).length > 0) {
            // Initialize Firebase services using the imported methods
            const appInstance = initializeApp(firebaseConfig);
            const dbInstance = getFirestore(appInstance);
            authInstance = getAuth(appInstance);
            setDb(dbInstance);
            setAuth(authInstance);

            // Sign-in logic
            const attemptAuth = async () => {
               try {
                  if (initialAuthToken) {
                     await signInWithCustomToken(
                        authInstance,
                        initialAuthToken
                     );
                  } else {
                     await signInAnonymously(authInstance);
                  }
               } catch (e) {
                 // console.error('Firebase Sign-In Failed:', e);
               }
            };
            attemptAuth();
         } else {
            console.warn(
               'Firebase config not available. Running in mock mode.'
            );
            setIsAuthReady(true);
         }
      } catch (e) {
         console.error('Firebase initialization failed:', e);
      }

      // Auth state listener
      const unsubscribe = authInstance
         ? onAuthStateChanged(authInstance, (user) => {
              setUserId(user ? user.uid : null);
              setIsAuthReady(true);
           })
         : () => {};

      // Cleanup
      return () => unsubscribe();
   }, [initialAuthToken, JSON.stringify(firebaseConfig)]); // Used JSON.stringify for object comparison in dependency array

   // Simple routing function
   const navigate = useCallback((path) => {
      setCurrentPath(path);
      window.scrollTo(0, 0); // Scroll to top on navigation
   }, []);

   // 2. Render the page based on the current path
   const renderPage = () => {
      switch (currentPath) {
         case '/':
            return <HomePage navigate={navigate} />;
         case '/courses':
            return <CoursesPage />;
         case '/pricing':
            return <PricingPage />;
         case '/about':
            return <AboutPage />;
         default:
            return <NotFound />;
      }
   };

   const currencyContextValue = {
      currency,
      conversionRate,
      isLoading,
      error,
   };

   return (
      <CurrencyContext.Provider value={currencyContextValue}>
         <div className="flex flex-col min-h-screen bg-white font-inter">
            <Navbar
               navigate={navigate}
               currency={currency}
               setCurrency={setCurrency}
               isAuthReady={isAuthReady}
               userId={userId}
            />
            <main className="flex-grow">{renderPage()}</main>
            <Footer />
         </div>
      </CurrencyContext.Provider>
   );
};

export default App;

// --- Navbar Component (Handles routing and highlighting) ---
const Navbar = ({ links }) => {
   const [isOpen, setIsOpen] = useState(false);
   const location = useLocation();

   const toggleMenu = () => {
      setIsOpen(!isOpen);
   };

   useEffect(() => {
      // Close menu when route changes
      setIsOpen(false);
   }, [location]);

   return (
      <nav className="fixed w-full z-30 bg-gray-900/90 backdrop-blur-sm shadow-xl border-b border-gray-800">
         <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
            <div className="flex justify-between items-center h-16">
               {/* Brand Logo and Name */}
               <Link to="/" className="flex items-center space-x-2">
                  <Code className="w-8 h-8 text-academy-accent" />
                  <span className="text-2xl font-bold text-white">
                     Davidson Tech Academy
                  </span>
               </Link>

               {/* Desktop Menu - All navigation links */}
               <div className="hidden md:flex items-center space-x-6">
                  {links.map((link) => (
                     <Link
                        key={link.path}
                        to={link.path}
                        className={`text-sm font-medium transition duration-150 ease-in-out px-3 py-2 rounded-lg 
                                    ${
                                       location.pathname === link.path
                                          ? 'text-academy-accent bg-gray-800 shadow-inner'
                                          : 'text-gray-300 hover:text-white hover:bg-gray-800'
                                    }`}
                     >
                        {link.name}
                     </Link>
                  ))}

                  {/* Enrollment/Sign-In Button */}
                  <Link
                     to="/enroll"
                     className="inline-flex items-center justify-center h-10 px-6 font-bold text-gray-900 bg-orange-600 rounded-lg hover:bg-orange-500 transition duration-300 shadow-lg ml-4"
                  >
                     Enroll / Sign In
                  </Link>
               </div>

               {/* Mobile Menu Button */}
               <div className="md:hidden flex items-center">
                  <button
                     onClick={toggleMenu}
                     className="inline-flex items-center justify-center p-2 rounded-md text-gray-400 hover:text-white hover:bg-gray-700 focus:outline-none focus:ring-2 focus:ring-inset focus:ring-academy-accent"
                  >
                     {isOpen ? (
                        <X className="block h-6 w-6" />
                     ) : (
                        <Menu className="block h-6 w-6" />
                     )}
                  </button>
               </div>
            </div>
         </div>

         {/* Mobile Menu Content */}
         {isOpen && (
            <div className="md:hidden">
               <div className="px-2 pt-2 pb-3 space-y-1 sm:px-3">
                  {links.map((link) => (
                     <Link
                        key={link.path}
                        to={link.path}
                        className={`block px-3 py-2 rounded-md text-base font-medium 
                                    ${
                                       location.pathname === link.path
                                          ? 'text-academy-accent bg-gray-800'
                                          : 'text-gray-300 hover:text-white hover:bg-gray-800'
                                    }`}
                     >
                        {link.name}
                     </Link>
                  ))}
                  <Link
                     to="/enroll"
                     className="block px-3 py-2 rounded-md text-base font-bold text-gray-900 bg-orange-600 hover:bg-orange-500 transition duration-300 mt-2"
                  >
                     Enroll / Sign In
                  </Link>
               </div>
            </div>
         )}
      </nav>
   );
};
